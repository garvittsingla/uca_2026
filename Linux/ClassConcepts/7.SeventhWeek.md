a new memory will be created for a in both the process and a new process table will be created .nice!
```c
#include <stdio.h>
#include <unistd.h>

int main() {
  int a = 10;
  pid_t pid = fork();
  if (pid == 0) {
    printf("In child process %d\n", a); //10 (different a made in memory)

  } else {
    a++;
    printf("Updated value of a is %d\n", a); //11 same a in memory
    sleep(5)
  }
}
```
